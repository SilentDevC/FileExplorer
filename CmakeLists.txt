cmake_minimum_required(VERSION 3.12)
project(FileExplorer LANGUAGES CXX)

# Set C++ standard
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find Boost package (Header-Only Boost components do not require COMPONENTS parameter)
find_package(Boost REQUIRED)

# Add executable and list source files
add_executable(FileExplorer
    CommandProcess.cpp
    ErrorLogger.cpp
    FileExplorer.cpp
    JsonHandlers.cpp
    main.cpp
)

# Link Boost headers and system thread library to the target
target_link_libraries(FileExplorer PRIVATE Boost::boost)

# Enable compiler warnings
if(MSVC)
    target_compile_options(FileExplorer PRIVATE /W4)
else()
    target_compile_options(FileExplorer PRIVATE -Wall -Wextra)
endif()
